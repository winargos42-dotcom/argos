        """Отправляет ETH на указанный адрес."""

            invest = self.get_balance() - 0.001  # оставляем немного на газ

        new_private_key = self._generate_new_key()

    def _generate_new_key(self):
        """Генерирует новый закрытый ключ (используя квантовую случайность)."""
        # Используем квантовый оракул для истинной случайности
        bits = self.quantum.generate_bits(256)
        # Преобразуем биты в байты (32 байта = 256 бит)
        key_bytes = int(bits, 2).to_bytes(32, byteorder='big')
        # Убеждаемся, что ключ валиден (Ethereum принимает любой 32-байтовый)
        return '0x' + key_bytes.hex()
